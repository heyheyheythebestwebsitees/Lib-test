--[[
	Prism UI — a from-scratch Roblox menu kit

	One file. Drop it in and call Prism.new() to get a window; every control
	lives under that window's Page objects. Nothing here depends on the
	executor besides optional clipboard/toggle helpers.

	Feature set: themed window, sidebar with sections, sub-tabs, grouped
	controls (toggle / slider / cycle / color / keybind / multiselect / list /
	button / note / info), search, Simple vs Advanced mode, tooltips, right-
	click keybind popup, animated selection pill and sub-tab thumb.

	Prism is a UI kit only — it does not read files, does not touch the game.
]]

local function tween(obj, info, props)
	local t = tweenService:Create(obj, info, props)
	t:Play()
	return t
end
Prism.tween = tween

local order = 0
local function nextOrder()
	order += 1
	return order
end
Prism.nextOrder = nextOrder

local function corner(parent, radius)
	local c = Instance.new("UICorner")
	c.CornerRadius = UDim.new(0, radius)
	c.Parent = parent
	return c
end

local function stroke(parent, color, transparency, thickness)
	local s = Instance.new("UIStroke")
	s.Color = color or COLORS.edge
	s.Thickness = thickness or 1
	s.Transparency = transparency or 0
	s.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
	s.LineJoinMode = Enum.LineJoinMode.Round
	s.Parent = parent
	return s
end

local function gradient(parent, c1, c2, rotation)
	local g = Instance.new("UIGradient")
	g.Color = ColorSequence.new(c1, c2)
	g.Rotation = rotation or 0
	g.Parent = parent
	return g
end

local function vlist(parent, pad, horizontal)
	local l = Instance.new("UIListLayout")
	l.SortOrder = Enum.SortOrder.LayoutOrder
	l.Padding = UDim.new(0, pad or 0)
	if horizontal then
		l.FillDirection = Enum.FillDirection.Horizontal
	end
	l.Parent = parent
	return l
end

local function padding(parent, t, r, b, l)
	local u = Instance.new("UIPadding")
	u.PaddingTop = UDim.new(0, t or 0)
	u.PaddingRight = UDim.new(0, r or 0)
	u.PaddingBottom = UDim.new(0, b or 0)
	u.PaddingLeft = UDim.new(0, l or 0)
	u.Parent = parent
	return u
end

local function setFont(label, font)
	if typeof(font) == "Font" then
		label.FontFace = font
	elseif typeof(font) == "EnumItem" then
		label.Font = font
	else
		label.FontFace = FONT.body
	end
end

local function label(parent, text, size, color, font)
	local t = Instance.new("TextLabel")
	t.BackgroundTransparency = 1
	setFont(t, font or FONT.body)
	t.TextSize = size or M.title
	t.TextColor3 = color or COLORS.text
	t.TextXAlignment = Enum.TextXAlignment.Left
	t.Text = text or ""
	t.Parent = parent
	return t
end

local function frame(parent, bg)
	local f = Instance.new("Frame")
	f.BorderSizePixel = 0
	if bg then
		f.BackgroundColor3 = bg
	else
		f.BackgroundTransparency = 1
	end
	if parent then
		f.Parent = parent
	end
	return f
end

local function button(parent, text, size, color, bg)
	local b = Instance.new("TextButton")
	b.BackgroundColor3 = bg or COLORS.surface
	b.BorderSizePixel = 0
	setFont(b, FONT.medium)
	b.TextSize = size or M.value
	b.TextColor3 = color or COLORS.dim
	b.Text = text or ""
	b.AutoButtonColor = false
	b.Parent = parent
	return b
end

local function hex(c)
	return ("#%02x%02x%02x"):format(
		math.floor(c.R * 255 + 0.5),
		math.floor(c.G * 255 + 0.5),
		math.floor(c.B * 255 + 0.5)
	)
end
Prism.hex = hex

local function textWidth(text, size, font)
	local ok, bounds = pcall(textService.GetTextSize, textService, tostring(text or ""), size or M.value, font or Enum.Font.BuilderSansBold, Vector2.new(2000, 200))
	if ok and bounds then
		return math.ceil(bounds.X)
	end
	return #tostring(text or "") * (size or M.value) * 0.6
end
Prism.textWidth = textWidth

-- wrapped-text height for a given width
local function textHeight(text, size, width)
	local ok, bounds = pcall(textService.GetTextSize, textService, tostring(text or ""), size or M.desc, Enum.Font.BuilderSans, Vector2.new(math.max(width, 40), 4000))
	if ok and bounds then
		return math.ceil(bounds.Y)
	end
	return size or M.desc
end
Prism.textHeight = textHeight

-- round a number to a step, stripping float dust
local function roundToStep(value, step)
	if not step or step <= 0 then
		return value
	end
	local digits = math.max(0, math.ceil(-math.log10(step)))
	local pow = 10 ^ digits
	return math.floor(value / step + 0.5) * step
end

-- parse a typed slider value ("0.3", "x0.3", "15k", "20%", "off")
local function parseNumber(input)
	if type(input) ~= "string" then
		return nil
	end
	local s = string.gsub(string.lower(string.gsub(input, "×", "")), "%s+", "")
	s = if string.match(s, "^%-?%d+,%d%d%d") then string.gsub(s, ",", "") else string.gsub(s, ",", ".")
	if s == "off" or s == "none" or s == "never" then
		return 0
	end
	local num, suffix = string.match(s, "^x?([%-]?%d*%.?%d+)(%a*)")
	local n = tonumber(num)
	if n == nil or n ~= n then
		return nil
	end
	if suffix == "k" then
		n *= 1000
	end
	return n
end
Prism.parseNumber = parseNumber

-- clip a value to [lo, hi]
local function clamp(v, lo, hi)
	return math.clamp(v, lo, hi)
end

-- §6 animated liquid indicator

-- a moving pill whose two edges ride NumberValues; `apply(a, b)` paints it
function Prism.liquid(holder, apply)
	local L = { tw = {}, placed = false }
	L.a = Instance.new("NumberValue")
	L.b = Instance.new("NumberValue")
	L.a.Parent = holder
	L.b.Parent = holder

	local function paint()
		apply(L.a.Value, L.b.Value)
	end
	L.a.Changed:Connect(paint)
	L.b.Changed:Connect(paint)

	function L.to(a, b, animate)
		for _, t in ipairs(L.tw) do
			t:Cancel()
		end
		table.clear(L.tw)
		if not animate or not L.placed then
			L.placed = true
			L.a.Value = a
			L.b.Value = b
			paint()
			return
		end
		local forward = a > L.a.Value
		table.insert(L.tw, tween(L.a, if forward then TW.trail else TW.lead, { Value = a }))
		table.insert(L.tw, tween(L.b, if forward then TW.lead else TW.trail, { Value = b }))
	end
	return L
end

-- §7 drag manager
-------------------------------------------------------------------------------
Prism.drag = nil
Prism.dragInstalled = false

function Prism.beginDrag(onMove, onEnd)
	local prev = Prism.drag
	if prev and prev.onEnd then
		pcall(prev.onEnd)
	end
	Prism.drag = { onMove = onMove, onEnd = onEnd }
	if not Prism.dragInstalled then
		Prism.dragInstalled = true
		Prism.conn(userInputService.InputChanged:Connect(function(input)
			local d = Prism.drag
			if d and input.UserInputType == Enum.UserInputType.MouseMovement then
				d.onMove(Vector2.new(input.Position.X, input.Position.Y))
			end
		end))
		Prism.conn(userInputService.InputEnded:Connect(function(input)
			local d = Prism.drag
			if d and input.UserInputType == Enum.UserInputType.MouseButton1 then
				Prism.drag = nil
				if d.onEnd then
					d.onEnd()
				end
			end
		end))
	end
end

function Prism.conn(connection)
	table.insert(Prism.conns, connection)
	return connection
end

-- a group is a titled card with a hairline header
function Prism.group(parent, title, options)
	options = options or {}
	local groupFrame = frame(parent)
	groupFrame.Name = "Group_" .. tostring(title)
	groupFrame.AutomaticSize = Enum.AutomaticSize.Y
	groupFrame.Size = UDim2.new(1, 0, 0, 0)
	groupFrame.LayoutOrder = nextOrder()
	vlist(groupFrame, 0)
	padding(groupFrame, 0, 0, 14, 0)

	local card = frame(groupFrame, COLORS.surface:Lerp(COLORS.bg, 0.5))
	card.Name = "Card"
	card.AutomaticSize = Enum.AutomaticSize.Y
	card.Size = UDim2.new(1, 0, 0, 0)
	corner(card, 10)
	stroke(card, COLORS.hairline, 0)
	vlist(card, 0)
	padding(card, 0, 12, 8, 12)

	local head = frame(card)
	head.Name = "Head"
	head.Size = UDim2.new(1, 0, 0, 30)
	head.LayoutOrder = nextOrder()
	local titleLabel = label(head, string.lower(title), M.small + 1, COLORS.dim, FONT.semibold)
	titleLabel.Size = UDim2.new(1, 0, 1, 0)
	local hairline = frame(head, COLORS.hairline)
	hairline.Size = UDim2.new(1, 24, 0, 1)
	hairline.Position = UDim2.new(0, -12, 1, -1)
	local gap = frame(card)
	gap.Size = UDim2.new(1, 0, 0, 4)
	gap.LayoutOrder = nextOrder()

	local g = {
		frame = groupFrame,
		rows = {},
		title = title,
		advanced = options.advanced == true,
		simple = options.simple == true,
	}
	table.insert(Prism.groups, g)

	-- a bare row with an optional label and desc; sub-options parent into the group's current sub state
	local function row(height, text, rowOpts)
		rowOpts = rowOpts or {}
		local r = frame(card)
		r.Size = UDim2.new(1, 0, 0, height)
		r.LayoutOrder = nextOrder()
		local entry = {
			frame = r,
			text = string.lower(tostring(text or "")),
			desc = rowOpts.desc,
			advanced = if rowOpts.advanced ~= nil then rowOpts.advanced else g.advanced,
			base = height,
			isNote = rowOpts.isNote == true,
		}
		if g.subParent then
			entry.parent = g.subParent
			entry.advanced = false
			local pad = Instance.new("UIPadding")
			pad.PaddingLeft = UDim.new(0, 16)
			pad.Parent = r
			local rail = frame(r, COLORS.hairline)
			rail.Name = "SubRail"
			rail.Size = UDim2.new(0, 1, 1, -6)
			rail.Position = UDim2.new(0, -10, 0, 3)
			r.Visible = false
		end
		table.insert(Prism.rows, entry)
		table.insert(g.rows, entry)
		return r
	end
	g.row = row

	-- the row's label plus a help icon whose tooltip carries the description
	local function rowLabel(r, text, rowOpts, reserve)
		local l = label(r, string.lower(text), M.title, COLORS.text, FONT.body)
		l.Name = "Label"
		l.Size = UDim2.new(1, -((reserve or 0) + 24), 1, 0)
		l.TextTruncate = Enum.TextTruncate.AtEnd
		local helpText = if type(rowOpts.desc) == "string" and rowOpts.desc ~= "" then rowOpts.desc else nil
		if helpText then
			local hint = frame(r)
			hint.Name = "Hint"
			hint.Size = UDim2.fromOffset(16, 16)
			hint.AnchorPoint = Vector2.new(0, 0.5)
			hint.Position = UDim2.new(0, 0, 0.5, 0)
			local ic = label(hint, "?", M.small, COLORS.faint, FONT.semibold)
			ic.Size = UDim2.fromScale(1, 1)
			ic.TextXAlignment = Enum.TextXAlignment.Center
			local function place()
				hint.Position = UDim2.new(0, math.floor(l.TextBounds.X) + 6, 0.5, 0)
			end
			place()
			l:GetPropertyChangedSignal("TextBounds"):Connect(place)
			Prism.tip(hint, helpText)
			Prism.tip(l, helpText)
		end
		return l
	end
	g.rowLabel = rowLabel

	-- ── toggle ─────────────────────────────────────────────────────────────
	function g.toggle(_, text, key, toggleOpts)
		toggleOpts = toggleOpts or {}
		toggleOpts.key = key
		local r = row(M.row, text, toggleOpts)
		local l = rowLabel(r, text, toggleOpts, M.pillW + 8)

		-- the switch is a rounded track with a knob that slides on/off
		local pill = button(r, "", 10, COLORS.text, COLORS.off)
		pill.Name = "Switch"
		pill.BackgroundTransparency = 1
		pill.Size = UDim2.new(0, M.pillW, 0, M.pillH)
		pill.AnchorPoint = Vector2.new(1, 0.5)
		pill.Position = UDim2.new(1, 0, 0.5, 0)

		local track = frame(pill, COLORS.off)
		track.Size = UDim2.fromScale(1, 1)
		corner(track, M.pillH // 2)

		local fill = frame(pill, Color3.new(1, 1, 1))
		fill.Size = UDim2.fromScale(1, 1)
		fill.BackgroundTransparency = 1
		corner(fill, M.pillH // 2)
		gradient(fill, COLORS.accent2, COLORS.accent)

		local knob = frame(pill, COLORS.knobOff)
		knob.Size = UDim2.new(0, M.knob, 0, M.knob)
		knob.AnchorPoint = Vector2.new(0, 0.5)
		knob.Position = UDim2.new(0, 2, 0.5, 0)
		corner(knob, M.knob // 2)

		local bindTag = label(r, "", M.small, COLORS.muted, FONT.medium)
		bindTag.Name = "Bind"
		bindTag.Size = UDim2.new(0, 30, 0, 18)
		bindTag.AnchorPoint = Vector2.new(1, 0.5)
		bindTag.Position = UDim2.new(1, -(M.pillW + 8), 0.5, 0)
		bindTag.BackgroundColor3 = COLORS.side
		bindTag.BackgroundTransparency = 0
		bindTag.TextXAlignment = Enum.TextXAlignment.Center
		bindTag.Visible = false
		corner(bindTag, 4)
		stroke(bindTag, COLORS.edge, 0)

		local lastShown = nil
		local function refresh(animate)
			local on = toggleOpts.state and toggleOpts.state() or false
			lastShown = on
			local knobGoal = if on then UDim2.new(1, -(M.knob + 2), 0.5, 0) else UDim2.new(0, 2, 0.5, 0)
			if animate then
				tween(knob, TW.toggle, { Position = knobGoal, BackgroundColor3 = if on then Color3.new(1, 1, 1) else COLORS.knobOff })
				tween(fill, TW.toggle, { BackgroundTransparency = if on then 0 else 1 })
				tween(l, TW.toggle, { TextColor3 = if on then COLORS.text else COLORS.dim })
			else
				knob.Position = knobGoal
				knob.BackgroundColor3 = if on then Color3.new(1, 1, 1) else COLORS.knobOff
				fill.BackgroundTransparency = if on then 0 else 1
				l.TextColor3 = if on then COLORS.text else COLORS.dim
			end
		end
		refresh(false)

		table.insert(Prism.sync, function()
			local on = toggleOpts.state and toggleOpts.state() or false
			if on ~= lastShown then
				refresh(true)
			end
		end)

		local function refreshBind()
			local b = Prism.binds and Prism.binds[key]
			if type(b) == "table" and ((b.key and b.key ~= "") or b.mode == "Always" or b.mode == "Never") then
				local mode = b.mode or "Toggle"
				bindTag.Text = if mode == "Toggle" then Prism.keyLabel(b.key) elseif mode == "Hold" then Prism.keyLabel(b.key) .. " hold" else string.lower(mode)
				bindTag.Size = UDim2.new(0, math.max(24, textWidth(bindTag.Text, M.small) + 14), 0, 18)
				bindTag.Visible = true
			else
				bindTag.Text = ""
				bindTag.Visible = false
			end
		end
		refreshBind()
		Prism.bindTags[key] = refreshBind

		pill.MouseEnter:Connect(function()
			tween(knob, TW.hover, { BackgroundColor3 = if (toggleOpts.state and toggleOpts.state()) then Color3.new(1, 1, 1) else COLORS.dim })
		end)
		pill.MouseLeave:Connect(function()
			refresh(false)
		end)

		pill.MouseButton1Click:Connect(function()
			if toggleOpts.onChange then
				toggleOpts.onChange(not (toggleOpts.state and toggleOpts.state() or false))
			end
			refresh(true)
		end)

		-- right-click → keybind popup
		r.Active = true
		local function onRightClick(input)
			if input.UserInputType == Enum.UserInputType.MouseButton2 then
				Prism.openBindPopup(key, text, userInputService:GetMouseLocation())
			end
		end
		r.InputBegan:Connect(onRightClick)
		pill.InputBegan:Connect(onRightClick)

		-- sub-options: rows that only show while this toggle's three-line button is open
		local subState = nil
		local expandBtn = nil
		if type(toggleOpts.sub) == "function" then
			subState = { open = false, when = toggleOpts.subWhen }
			local eb = button(r, "", M.small, COLORS.text, COLORS.card)
			eb.Name = "Expand"
			eb.BackgroundTransparency = 1
			eb.Size = UDim2.fromOffset(20, 20)
			eb.AnchorPoint = Vector2.new(1, 0.5)
			eb.Position = UDim2.new(1, -(M.pillW + 12), 0.5, 0)
			corner(eb, 5)
			local ebLabel = label(eb, "≡", M.title, COLORS.muted, FONT.semibold)
			ebLabel.Size = UDim2.fromScale(1, 1)
			ebLabel.TextXAlignment = Enum.TextXAlignment.Center
			expandBtn = eb
			Prism.tip(eb, "More options for " .. tostring(text))

			local prevParent = g.subParent
			g.subParent = subState
			local okSub, errSub = pcall(toggleOpts.sub, g)
			g.subParent = prevParent
			if not okSub then
				warn("prism: sub-options of " .. tostring(text) .. " failed: " .. tostring(errSub))
			end

			eb.MouseButton1Click:Connect(function()
				local opening = not subState.open
				if opening then
					subState.open = true
					ebLabel.TextColor3 = COLORS.accent
				else
					subState.open = false
					ebLabel.TextColor3 = COLORS.muted
				end
				Prism.refilter()
			end)
		end

		return r
	end

	-- ── slider ─────────────────────────────────────────────────────────────
	function g.slider(_, text, key, min, max, step, sliderOpts)
		sliderOpts = sliderOpts or {}
		sliderOpts.key = key
		local r = row(M.row, text, sliderOpts)
		local trackW = sliderOpts.trackW or M.trackW
		rowLabel(r, text, sliderOpts, trackW + 10 + M.valueW)

		local valueBtn = button(r, "", M.value, COLORS.text, COLORS.surface)
		valueBtn.Name = "Value"
		valueBtn.BackgroundTransparency = 1
		valueBtn.Size = UDim2.new(0, M.valueW, 1, 0)
		valueBtn.AnchorPoint = Vector2.new(1, 0)
		valueBtn.Position = UDim2.new(1, 0, 0, 0)
		valueBtn.TextXAlignment = Enum.TextXAlignment.Right

		local rail = button(r, "", 10, COLORS.text, COLORS.off)
		rail.Name = "Rail"
		rail.BackgroundTransparency = 1
		rail.Size = UDim2.new(0, trackW, 0, 16)
		rail.AnchorPoint = Vector2.new(1, 0.5)
		rail.Position = UDim2.new(1, -(M.valueW + 10), 0.5, 0)

		local track = frame(rail, COLORS.off)
		track.Size = UDim2.new(1, 0, 0, 4)
		track.AnchorPoint = Vector2.new(0, 0.5)
		track.Position = UDim2.new(0, 0, 0.5, 0)
		corner(track, 2)

		local fill = frame(track, Color3.new(1, 1, 1))
		fill.Size = UDim2.new(0, 0, 1, 0)
		corner(fill, 2)
		gradient(fill, COLORS.accent2, COLORS.accent)

		local knob = frame(track, Color3.new(1, 1, 1))
		knob.Size = UDim2.new(0, 10, 0, 10)
		knob.AnchorPoint = Vector2.new(0.5, 0.5)
		knob.Position = UDim2.new(0, 0, 0.5, 0)
		corner(knob, 5)
		local ring = stroke(knob, COLORS.accent, 1, 2)

		local function fmt(n)
			if sliderOpts.fmt then
				return sliderOpts.fmt(n)
			end
			return tostring(math.floor(n * 100 + 0.5) / 100)
		end
		local unit = if type(sliderOpts.unit) == "string" and sliderOpts.unit ~= "" then sliderOpts.unit else nil
		local shownText = ""

		local function refresh(animateKnob)
			local n = tonumber(sliderOpts.state and sliderOpts.state() or min) or min
			local rel = clamp((n - min) / math.max(max - min, 1e-6), 0, 1)
			local kp = UDim2.new(rel, 0, 0.5, 0)
			if animateKnob then
				tween(knob, TW.knob, { Position = kp })
				tween(fill, TW.knob, { Size = UDim2.new(rel, 0, 1, 0) })
			else
				knob.Position = kp
				fill.Size = UDim2.new(rel, 0, 1, 0)
			end
			shownText = fmt(n)
			valueBtn.Text = if unit then shownText .. " " .. unit else shownText
		end

		local function setRing(on)
			tween(ring, TW.hover, { Transparency = if on then 0.7 else 1 })
		end

		rail.MouseEnter:Connect(function()
			setRing(true)
		end)
		rail.MouseLeave:Connect(function()
			if not (Prism.drag and Prism.drag.owner == rail) then
				setRing(false)
			end
		end)

		local function setFromX(x, dragging)
			local rel = clamp((x - rail.AbsolutePosition.X) / math.max(rail.AbsoluteSize.X, 1), 0, 1)
			local raw = min + (max - min) * rel
			local n = clamp(roundToStep(raw, step), min, max)
			if sliderOpts.onChange then
				sliderOpts.onChange(n)
			end
			refresh(not dragging)
		end

		refresh(false)
		table.insert(Prism.sync, function()
			local n = tonumber(sliderOpts.state and sliderOpts.state() or min) or min
			if fmt(n) ~= shownText then
				refresh(true)
			end
		end)

		rail.InputBegan:Connect(function(input)
			if input.UserInputType == Enum.UserInputType.MouseButton1 then
				setFromX(input.Position.X, true)
				setRing(true)
				Prism.beginDrag(function(pos)
					setFromX(pos.X, true)
				end, function()
					setRing(false)
					refresh(true)
				end)
				if Prism.drag then
					Prism.drag.owner = rail
				end
			end
		end)

		-- click the number → type a value
		local editing = false
		valueBtn.MouseButton1Click:Connect(function()
			if editing then
				return
			end
			editing = true
			local box = Instance.new("TextBox")
			box.Size = UDim2.new(0, math.max(M.valueW + 26, 68), 0, 22)
			box.AnchorPoint = Vector2.new(1, 0.5)
			box.Position = UDim2.new(1, 2, 0.5, 0)
			box.BackgroundColor3 = COLORS.surface
			box.BorderSizePixel = 0
			setFont(box, FONT.medium)
			box.TextSize = M.value
			box.TextColor3 = COLORS.text
			box.Text = tostring(tonumber(sliderOpts.state and sliderOpts.state() or min) or min)
			box.ClearTextOnFocus = false
			box.TextXAlignment = Enum.TextXAlignment.Right
			box.ZIndex = 5
			box.Parent = r
			corner(box, 5)
			stroke(box, COLORS.accent, 0.2)
			padding(box, 0, 6, 0, 6)
			box:CaptureFocus()
			box.FocusLost:Connect(function(enterPressed)
				editing = false
				local n = if enterPressed then parseNumber(box.Text) else nil
				box:Destroy()
				if n == nil then
					return
				end
				n = clamp(roundToStep(n, step), min, max)
				if sliderOpts.onChange then
					sliderOpts.onChange(n)
				end
				refresh(true)
			end)
		end)

		return r
	end

	-- ── button ─────────────────────────────────────────────────────────────
	function g.button(_, text, fn, tint, buttonOpts)
		buttonOpts = buttonOpts or {}
		local r = row(M.rowButton, text, buttonOpts)
		local base = tint or COLORS.surface
		local b = button(r, string.lower(text), M.value, tint or COLORS.dim, base)
		b.Size = UDim2.new(1, 0, 0, 26)
		b.AnchorPoint = Vector2.new(0, 0.5)
		b.Position = UDim2.new(0, 0, 0.5, 0)
		b.BackgroundTransparency = if tint then 0.86 else 0
		b.TextTruncate = Enum.TextTruncate.AtEnd
		corner(b, 6)
		local bs = stroke(b, tint or COLORS.edge, if tint then 0.6 else 0)

		b.MouseEnter:Connect(function()
			if tint then
				tween(b, TW.hover, { BackgroundTransparency = 0.78 })
			else
				tween(b, TW.hover, { BackgroundColor3 = COLORS.card, TextColor3 = COLORS.text })
			end
		end)
		b.MouseLeave:Connect(function()
			if tint then
				tween(b, TW.hover, { BackgroundTransparency = 0.86 })
			else
				tween(b, TW.hover, { BackgroundColor3 = COLORS.surface, TextColor3 = COLORS.dim })
			end
		end)
		b.MouseButton1Down:Connect(function()
			b.Position = UDim2.new(0, 0, 0.5, 1)
			bs.Transparency = 0
		end)
		b.MouseButton1Up:Connect(function()
			b.Position = UDim2.new(0, 0, 0.5, 0)
			bs.Transparency = if tint then 0.6 else 0
		end)
		b.MouseButton1Click:Connect(function()
			local ok, err = pcall(fn)
			if not ok then
				warn("prism button " .. tostring(text) .. ": " .. tostring(err))
			end
		end)

		if type(buttonOpts.desc) == "string" and buttonOpts.desc ~= "" then
			Prism.tip(b, buttonOpts.desc)
		end
		return b
	end

	function g.multiselect(_, text, key, options, multiOpts)
		multiOpts = multiOpts or {}

		local r = row(M.row, text, multiOpts)
		rowLabel(r, text, multiOpts, M.ctrlW)

		b.Name = "Multiselect"
		b.Size = UDim2.new(0, M.ctrlW, 0, 22)
		b.AnchorPoint = Vector2.new(1, 0.5)
		b.Position = UDim2.new(1, 0, 0.5, 0)
		corner(b, 5)

		local val = label(b, "", M.value, COLORS.text, FONT.body)

			bs.Color = COLORS.accent
			Prism.openDropdown(b, M.ctrlW, function(dropFrame)

					Prism.dropdownOption(dropFrame, 4 + (i - 1) * 24, string.lower(opt), selected[opt] == true, function()

							multiOpts.onChange(opt, not selected[opt])

	-- ── list (single select, with optional search) ─────────────────────────
	function g.list(_, text, listOpts)
		listOpts = listOpts or {}
		listOpts.isList = true
		local rowH = M.listRow
		local headerH = if listOpts.search ~= false then 52 else 24
		local wanted = math.ceil((listOpts.height or 200) * 1.1)
		local rowsFit = math.max(3, math.floor((wanted - headerH - 8) / rowH))
		local totalH = headerH + rowsFit * rowH + 8

		local r = row(totalH, text, listOpts)
		local title = label(r, string.lower(text), M.title, COLORS.text, FONT.body)
		title.Size = UDim2.new(1, -110, 0, 22)
		local countLabel = label(r, "", M.small, COLORS.muted, FONT.medium)
		countLabel.Size = UDim2.new(0, 100, 0, 22)
		countLabel.Position = UDim2.new(1, -100, 0, 0)
		countLabel.TextXAlignment = Enum.TextXAlignment.Right

		local top = 24
		local query = ""
		local refresh
		if listOpts.search ~= false then
			local box = Instance.new("TextBox")
			box.Size = UDim2.new(1, 0, 0, 24)
			box.Position = UDim2.new(0, 0, 0, 24)
			box.BackgroundColor3 = COLORS.surface
			box.BorderSizePixel = 0
			setFont(box, FONT.body)
			box.TextSize = M.value
			box.TextColor3 = COLORS.text
			box.PlaceholderText = "search"
			box.PlaceholderColor3 = COLORS.muted
			box.Text = ""
			box.ClearTextOnFocus = false
			box.TextXAlignment = Enum.TextXAlignment.Left
			box.Parent = r
			corner(box, 6)
			stroke(box, COLORS.edge, 0)
			padding(box, 0, 8, 0, 10)
			box:GetPropertyChangedSignal("Text"):Connect(function()
				query = string.lower(box.Text)
				if refresh then
					refresh()
				end
			end)
			top = 52
		end

		local scroller = Instance.new("ScrollingFrame")
		scroller.Size = UDim2.new(1, 0, 1, -top)
		scroller.Position = UDim2.new(0, 0, 0, top)
		scroller.BackgroundColor3 = COLORS.side
		scroller.BorderSizePixel = 0
		scroller.ScrollBarThickness = 4
		scroller.ScrollBarImageColor3 = COLORS.faint
		scroller.AutomaticCanvasSize = Enum.AutomaticSize.Y
		scroller.CanvasSize = UDim2.new(0, 0, 0, 0)
		scroller.ScrollingDirection = Enum.ScrollingDirection.Y
		scroller.Parent = r
		corner(scroller, 6)
		stroke(scroller, COLORS.edge, 0)
		vlist(scroller, 2)
		padding(scroller, 4, 6, 4, 4)

		local pool = {}
		local rowIds = {}
		local lastSelection = nil

		local function makeRow()
			local b = button(scroller, "", M.value, COLORS.dim, COLORS.accent)
			b.BackgroundTransparency = 1
			b.Size = UDim2.new(1, 0, 0, rowH - 2)
			b.TextXAlignment = Enum.TextXAlignment.Left
			b.TextTruncate = Enum.TextTruncate.AtEnd
			setFont(b, FONT.body)
			corner(b, 5)
			padding(b, 0, 8, 0, 8)
			local s = stroke(b, COLORS.accent, 1)
			local sub = label(b, "", M.small, COLORS.muted, FONT.medium)
			sub.Size = UDim2.new(0, 120, 1, 0)
			sub.Position = UDim2.new(1, -120, 0, 0)
			sub.TextXAlignment = Enum.TextXAlignment.Right
			local rec = { btn = b, sub = sub, stroke = s, selected = false }
			b.MouseEnter:Connect(function()
				if not rec.selected then
					tween(b, TW.hover, { BackgroundTransparency = 0.88 })
				end
			end)
			b.MouseLeave:Connect(function()
				if not rec.selected then
					tween(b, TW.hover, { BackgroundTransparency = 1 })
				end
			end)
			b.MouseButton1Click:Connect(function()
				local id = rowIds[b]
				if id ~= nil and listOpts.onSelect then
					local ok, err = pcall(listOpts.onSelect, id)
					if not ok then
						warn("prism list " .. tostring(text) .. ": " .. tostring(err))
					end
					refresh()
				end
			end)
			return rec
		end

		refresh = function()
			local items = if listOpts.items then listOpts.items() else {}
			local selection = if listOpts.selected then listOpts.selected() else nil
			lastSelection = selection
			local shown = 0
			for _, item in ipairs(items) do
				local lab = tostring(item.label or item.id)
				local matches = query == ""
					or string.find(string.lower(lab), query, 1, true)
					or (item.sub and string.find(string.lower(tostring(item.sub)), query, 1, true))
				if matches then
					shown += 1
					local rec = pool[shown]
					if not rec then
						rec = makeRow()
						pool[shown] = rec
					end
					local b = rec.btn
					rowIds[b] = item.id
					b.Visible = true
					b.LayoutOrder = shown
					b.Text = lab
					rec.sub.Text = tostring(item.sub or "")
					rec.sub.TextColor3 = item.color or COLORS.muted
					rec.selected = item.id == selection
					b.TextColor3 = if rec.selected then COLORS.text elseif item.dim then COLORS.muted else COLORS.dim
					b.BackgroundColor3 = COLORS.accent
					b.BackgroundTransparency = if rec.selected then 0.82 else 1
					rec.stroke.Transparency = if rec.selected then 0.35 else 1
				end
			end
			for i = shown + 1, #pool do
				pool[i].btn.Visible = false
			end
			countLabel.Text = if shown == #items then tostring(#items) else ("%d / %d"):format(shown, #items)
		end
		refresh()
		table.insert(Prism.sync, function()
			local selection = if listOpts.selected then listOpts.selected() else nil
			if selection ~= lastSelection then
				refresh()
			end
		end)
		return refresh
	end

	return g
end

local tipFrame = nil

function Prism.tip(target, text)
	if type(text) ~= "string" or text == "" then
		return
	end
	target.MouseEnter:Connect(function()
		if not Prism.gui then
			return
		end
		if not tipFrame or not tipFrame.Parent then
			local t = Instance.new("TextLabel")
			t.Name = "Tooltip"
			t.AutomaticSize = Enum.AutomaticSize.XY
			t.Size = UDim2.new(0, 0, 0, 0)
			t.BackgroundColor3 = COLORS.pop
			setFont(t, FONT.body)
			t.TextSize = M.desc
			t.TextColor3 = COLORS.text
			t.TextWrapped = true
			t.TextXAlignment = Enum.TextXAlignment.Left
			t.ZIndex = 200
			t.Visible = false
			t.Parent = Prism.gui
			corner(t, 6)
			stroke(t, COLORS.edge, 0)
			padding(t, 6, 10, 6, 10)
			local cap = Instance.new("UISizeConstraint")
			cap.MaxSize = Vector2.new(260, 400)
			cap.Parent = t
			tipFrame = t
		end
		local tf = tipFrame :: TextLabel
		tf.Text = text
		local m = userInputService:GetMouseLocation()
		local inset = if Prism.gui.IgnoreGuiInset then 0 else guiService:GetGuiInset().Y
		tf.Position = UDim2.fromOffset(m.X + 14, m.Y - inset + 10)
		tf.Visible = true
	end)
	target.MouseLeave:Connect(function()
		if tipFrame then
			tipFrame.Visible = false
		end
	end)
end

-- dropdown popover
Prism.popup = nil
Prism.popupHost = nil

function Prism.closeDropdown()
	local p = Prism.popup
	Prism.popup = nil
	if p then
		pcall(p.catcher.Destroy, p.catcher)
		pcall(p.frame.Destroy, p.frame)
		if p.onClose then
			pcall(p.onClose)
		end
	end
end

function Prism.openDropdown(anchor, width, build, onClose)
	Prism.closeDropdown()
	local host = Prism.popupHost or Prism.root
	if not host then
		return nil
	end
	local catcher = Instance.new("TextButton")
	catcher.Name = "DropdownCatcher"
	catcher.BackgroundTransparency = 1
	catcher.Text = ""
	catcher.AutoButtonColor = false
	catcher.Size = UDim2.new(1, 400, 1, 0)
	catcher.ZIndex = 58
	catcher.Parent = host

	local f = frame(host, COLORS.pop)
	f.Name = "Dropdown"
	f.ZIndex = 60

function Prism.keyLabel(name)

		return "—"

function Prism.closeBindPopup()
	local p = bindPopup
	bindPopup = nil
	if p then
		p:Destroy()
	end
end

function Prism.captureKey(callback)
	local finished = false
	local started = os.clock()
	local conn, heartbeat
	local function finish(name)
		if finished then
			return
		end
		finished = true
		if conn then
			conn:Disconnect()
		end
		if heartbeat then
			heartbeat:Disconnect()
		end
		callback(name)
	end
	conn = userInputService.InputBegan:Connect(function(input)
		if input.KeyCode == Enum.KeyCode.Escape or input.KeyCode == Enum.KeyCode.Backspace then
			finish(nil)
			return
		end
		if os.clock() - started < 0.2 and input.UserInputType == Enum.UserInputType.MouseButton1 then
			return
		end
		if input.KeyCode ~= Enum.KeyCode.Unknown then
			finish(input.KeyCode.Name)
		elseif input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.MouseButton2
			or input.UserInputType == Enum.UserInputType.MouseButton3 then
			finish(input.UserInputType.Name)
		end
	end)
	heartbeat = runService.Heartbeat:Connect(function()
		if os.clock() - started > 8 then
			finish(false)
		end
	end)
end

			task.defer(function()
				if token == animToken then
					placeNav(false)
				end
			end)
		else
			Prism.closeBindPopup()
			Prism.closeDropdown()
			Prism.slide(host, hostBase, 0, TW.close.Time, nil, true, function()
				if token == animToken and not Prism.visible then
					host.Visible = false
					gui.Enabled = false
				end
			end)
			return
		end
		Prism.slide(host, hostBase, 0, TW.open.Time, nil, false, nil)
	end

	Prism.root.Visible = true
	Prism.revealToken = 0

	function Prism.reveal(obj, base, dx, dy)
		Prism.settle(obj, base)
		local sheet = Prism.pageCurtain
		if not sheet then
			return
		end
		Prism.revealToken += 1
		local token = Prism.revealToken
		sheet.Position = UDim2.fromOffset(0, 0)
		sheet.BackgroundTransparency = 0.15
		sheet.Visible = true
		tween(sheet, TW.reveal, { BackgroundTransparency = 1 }).Completed:Connect(function(state)
			if state == Enum.PlaybackState.Completed and token == Prism.revealToken then
				sheet.Visible = false
			end
		end)
	end

	function Prism.settle(obj, base)
		if obj.Position ~= base then
			obj.Position = base
		end
	end

	-- page registry -------------------------------------------------------
	Prism.pages = {}
	Prism.subTabs = {}

	-- tab / sub-tab change -------------------------------------------------
	function Prism.showTab(name)
		local was = Prism.activeTab
		Prism.activeTab = name
		Prism.closeDropdown()
		local changed = was ~= name
		local animate = changed and was ~= "" and Prism.visible == true

		for n, page in pairs(Prism.pages) do
			local show = n == name
			if show and animate then
				Prism.reveal(page, UDim2.new())
			elseif changed then
				Prism.settle(page, UDim2.new())
			end
			page.Visible = show
		end
		for n, d in pairs(tabButtons) do
			if not changed or n == name or n == was or not animate then
				d.paint(n == name, animate)
			end
		end
		task.defer(function()
			placeNav(animate)
		end)
		if changed then
			Prism.syncNow(name)
		end
	end

	function Prism.showSub(tab, sub)
		local st = Prism.subTabs[tab]
		if not st then
			return
		end
		Prism.closeDropdown()
		local was = st.active
		st.active = sub
		local changed = was ~= sub
		if st.crumb then
			st.crumb.Text = string.lower((TAB_LABEL[tab] or tab) .. " / " .. sub)
		end
		local animate = changed and st.built == true and Prism.visible == true and Prism.activeTab == tab

		for n, fr in pairs(st.frames) do
			local show = n == sub
			if show and animate then
				Prism.reveal(fr, st.base)
			elseif changed then
				Prism.settle(fr, st.base)
			end
			fr.Visible = show
		end
		for n, b in pairs(st.btns) do
			local on = n == sub
			if animate then
				tween(b.lbl, TW.hover, { TextColor3 = if on then COLORS.text else COLORS.muted })
			else
				b.lbl.TextColor3 = if on then COLORS.text else COLORS.muted
			end
			setFont(b.lbl, if on then FONT.medium else FONT.body)
		end
		if animate then
			task.defer(function()
				Prism.placeSubThumb(tab, true)
			end)
		else
			Prism.placeSubThumb(tab, false)
		end
	end

	function Prism.placeSubThumb(tab, animate)
		local st = Prism.subTabs[tab]
		if not (st and st.liquid and st.track) then
			return
		end
		local b = st.btns[st.active]
		local btn = b and b.btn
		if not (btn and btn.Visible) then
			st.thumb.Visible = false
			return
		end
		local w = btn.AbsoluteSize.X
		if w < 1 then
			return
		end
		st.thumb.Visible = true
		local x = btn.AbsolutePosition.X - st.track.AbsolutePosition.X
		st.liquid.to(x, x + w, animate and Prism.visible == true)
	end

	-- search + filtering ----------------------------------------------------
	function Prism.search(text)
		Prism.searchText = string.lower(tostring(text or ""))
		Prism.refilter()
	end

	-- Simple mode whitelist: put your keys here. Simple shows only these.
	Prism.simpleKeys = {}

	function Prism.modeAllows(advanced)
		return Prism.uiMode == "Advanced" or not advanced
	end

	function Prism.refilter()
		for _, r in ipairs(Prism.rows) do
			local matchText = true
			if Prism.searchText ~= "" then
				matchText = string.find(string.lower(r.text), Prism.searchText, 1, true) ~= nil
					or (r.desc and string.find(string.lower(r.desc), Prism.searchText, 1, true) ~= nil)
			end
			local parentOpen = true
			if r.parent then
				parentOpen = r.parent.open == true or Prism.searchText ~= ""
					or (r.parent.when ~= nil and r.parent.when() == true)
			end
			r.frame.Visible = Prism.modeAllows(r.advanced) and matchText and parentOpen
		end
		for _, g in ipairs(Prism.groups) do
			local any = false
			for _, r in ipairs(g.rows) do
				if r.frame.Visible then
					any = true
					break
				end
			end
			g.frame.Visible = Prism.modeAllows(g.advanced) and (any or (#g.rows == 0 and Prism.searchText == ""))
		end
	end

	function Prism.syncNow(tab)
		for _, fn in ipairs(Prism.sync) do
			pcall(fn)
		end
	end

	-- dragging --------------------------------------------------------------
	local dragging, dragStart, startPos = false, Vector2.zero, UDim2.new()
	local dragConn = userInputService.InputChanged:Connect(function(input)
		if dragging and input.UserInputType == Enum.UserInputType.MouseMovement then
			local d = input.Position - dragStart
			host.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X, startPos.Y.Scale, startPos.Y.Offset + d.Y)
		end
	end)
	Prism.conn(dragConn)

	for _, handle in ipairs({ top, logo }) do
		handle.InputBegan:Connect(function(input)
			if input.UserInputType == Enum.UserInputType.MouseButton1 then
				dragging = true
				dragStart = input.Position
				startPos = host.Position
			end
		end)
		handle.InputEnded:Connect(function(input)
			if input.UserInputType == Enum.UserInputType.MouseButton1 then
				dragging = false
				hostBase = host.Position
			end
		end)
	end

	Prism.visible = true
	return Prism
end

-------------------------------------------------------------------------------
-- §11 animation helper for sliding the window
-------------------------------------------------------------------------------
function Prism.slide(obj, base, dy, dur, fade, closing, done)
	local info = TweenInfo.new(dur, Enum.EasingStyle.Quart, if closing then Enum.EasingDirection.In else Enum.EasingDirection.Out)
	obj.Position = if closing then base else base + UDim2.fromOffset(0, dy)
	local t = tweenService:Create(obj, info, { Position = if closing then base + UDim2.fromOffset(0, dy) else base })
	t.Completed:Connect(function(state)
		if state == Enum.PlaybackState.Completed then
			if closing and obj.Parent then
				obj.Position = base
			end
			if done then
				done()
			end
		end
	end)
	t:Play()
end
